import torch, numpy as np, segmentation_models_pytorch as smp
from torch.utils.data import Dataset, DataLoader
import torch.nn.functional as F

class Mars(Dataset):
    def __init__(self, hf, size=512): self.hf, self.size = hf, size
    def __len__(self): return len(self.hf)
    def __getitem__(self, i):
        img, lab = prepare(self.hf[i])                      # from your sheet
        x = torch.tensor(img).permute(2,0,1).float()/255
        x = F.interpolate(x[None], size=self.size, mode="bilinear")[0]
        y = torch.tensor(lab)[None,None].float()
        y = F.interpolate(y, size=self.size, mode="nearest")[0,0].long()
        if np.random.rand() < .5: x, y = x.flip(-1), y.flip(-1)   # same flip!
        x = (x - torch.tensor([.485,.456,.406])[:,None,None]) / torch.tensor([.229,.224,.225])[:,None,None]
        return x, y

model = smp.DeepLabV3Plus("resnet50", encoder_weights="imagenet", classes=4).cuda()
loss_fn = torch.nn.CrossEntropyLoss(weight=torch.tensor(weights, dtype=torch.float32).cuda(), ignore_index=255)
opt = torch.optim.AdamW(model.parameters(), lr=1e-4)

Why must the flip be applied to image AND mask?

Otherwise the labels point at the wrong pixels: the left-side sand would be labeled on the right.

Why is big rock IoU "jumpy" between runs?

Very few test pixels are big rock, so a few mistakes change the score a lot. Say so honestly.

Steps

elev = src.read(1, masked=True).filled(np.nan)
dzdy, dzdx = np.gradient(elev, src.res[0])            # res in METERS
slope = np.degrees(np.arctan(np.hypot(dzdx, dzdy)))
rgba = np.zeros((*slope.shape, 4), np.uint8)
rgba[slope < 15] = (46,139,74,140); rgba[(slope>=15)&(slope<20)] = (224,122,31,160)
rgba[slope >= 20] = (212,61,61,180); rgba[np.isnan(slope)] = 0
from PIL import Image; Image.fromarray(rgba).save("web/public/data/ov/slope.png")

Classic bug check

If almost everything is red, your pixel size is in degrees, not meters. If you see a giant hole, NoData wasn't masked.

Journal

Bilinear or nearest for the dune raster?

Nearest: it's a yes/no class map. Bilinear would invent "half-dune" values.

const reasons = [];
